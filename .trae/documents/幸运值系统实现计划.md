### 4.4 爱的力量技能实现

```typescript
// src/utils/lovePowerSkill.ts（新建）

export interface LovePowerResult {
  triggered: boolean;
  luckBonus: number;
  hpRestored: boolean;
  message: string;
}

export function checkLovePower(
  skillLevel: number,
  isPlayerDead: boolean,
  isPetDead: boolean
): LovePowerResult {
  // 技能等级1：20%概率，幸运值+10
  // 技能等级2：25%概率，幸运值+20
  if (skillLevel === 1 && Math.random() * 100 > 80) {
    return {
      triggered: true,
      luckBonus: 10,
      hpRestored: true,
      message: '爱的力量使你和你的幻兽生命值回复满，并且幸运值增加10点'
    };
  }
  if (skillLevel === 2 && Math.random() * 100 > 75) {
    return {
      triggered: true,
      luckBonus: 20,
      hpRestored: true,
      message: '爱的力量使你和你的幻兽生命值回复满，并且幸运值增加20点'
    };
  }
  return { triggered: false, luckBonus: 0, hpRestored: false, message: '' };
}
```
