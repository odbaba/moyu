# 幻兽升级逻辑重构计划

## 问题分析

当前幻兽升级逻辑存在以下问题：

1. **逻辑位置不合理**：升级逻辑分散在 `App.tsx` 中，不符合面向对象设计原则
2. **缺少统一的经验获取接口**：战斗结束和使用经验球的逻辑重复
3. **未实现完整的升级机制**：
   - 缺少等级限制检查（最高130级，不能超过人物等级+10）
   - 缺少经验值翻倍机制
   - 缺少顿悟机制
   - 缺少经验需求递增规则（参考文档：1-19级×1.2，20-50级×1.1，51-129级+hun）

## 重构目标

将幻兽升级逻辑封装到幻兽对象中，提供统一的 `gainExperience` 方法：

1. **封装升级逻辑**：在 `petGenerator.ts` 中创建 `gainExperience` 函数
2. **实现完整升级机制**：包括等级限制、经验翻倍、升级循环、顿悟等
3. **统一调用接口**：战斗结束和使用经验球时调用同一方法
4. **符合文档规范**：严格按照 reference 文档实现

## 详细实现规范

### 经验获取函数实现

```typescript
/**
 * 幻兽获得经验并处理升级
 * 参考文档：reference/docs/project_docs/02.1_幻兽升级经验系统.md
 * 
 * @param pet 幻兽对象
 * @param expAmount 获得的经验值（基础值，会自动翻倍）
 * @param playerLevel 玩家等级（用于等级限制检查）
 * @returns 更新后的幻兽数据和升级信息
 */
export function gainExperience(
  pet: Pet,
  expAmount: number,
  playerLevel: number
): {
  pet: Pet;
  leveledUp: boolean;
  message?: string;
} {
  // 1. 等级上限检查
  if (pet.dj >= 130) {
    return {
      pet: { ...pet, jy: 0 },
      leveledUp: false,
      message: "幻兽等级已满，无法再获得经验值了。"
    };
  }

  // 2. 人物等级关联检查
  if (pet.dj >= playerLevel + 10) {
    return {
      pet,
      leveledUp: false,
      message: "幻兽等级已高于人物的10级，无法再获得经验值了。"
    };
  }

  // 3. 幻兽获得双倍经验
  const doubledExp = expAmount * 2;
  let newExp = pet.jy + doubledExp;
  let newLevel = pet.dj;
  let newMaxExp = pet.mjy;
  let newHun = pet.hun;
  let newPredj = pet.predj;
  let leveledUp = false;
  let message: string | undefined;

  // 4. 升级循环处理
  while (newExp >= newMaxExp && newLevel < 130) {
    newExp -= newMaxExp;
    newLevel++;
    leveledUp = true;

    // 更新 predj（幻化前等级记录）
    if (newPredj < newLevel) {
      newPredj = newLevel;
    }

    // 根据等级计算新的 mjy
    if (newLevel < 20) {
      newMaxExp = Math.round(newMaxExp * 1.2);
    } else if (newLevel <= 50) {
      newMaxExp = Math.round(newMaxExp * 1.1);
    } else if (newLevel < 130) {
      newMaxExp += newHun;
    }

    // 50级特殊处理
    if (newLevel === 50) {
      newHun = Math.round(newMaxExp * 0.2);

      // 顿悟机制
      if (pet.predj > newLevel) {
        newLevel = pet.predj;
        newMaxExp = pet.premjy;
        newExp = pet.prejy;
        message = `幻兽在升级中顿悟了，等级立即升到幻化转世之前的等级${newLevel}级`;
        break;
      } else {
        message = "你的幻兽达到50级了，可以到幻化大师那里进行幻化。";
      }
    }
  }

  // 5. 如果升级了，更新属性
  let updatedPet: Pet;
  if (newLevel !== pet.dj) {
    updatedPet = upgradePetLevel(
      {
        ...pet,
        jy: newExp,
        mjy: newMaxExp,
        hun: newHun,
        predj: newPredj,
      },
      newLevel
    );
  } else {
    updatedPet = {
      ...pet,
      jy: newExp,
      mjy: newMaxExp,
      hun: newHun,
      predj: newPredj,
    };
  }

  return {
    pet: updatedPet,
    leveledUp,
    message,
  };
}
```
