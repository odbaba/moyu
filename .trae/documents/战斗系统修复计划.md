# 战斗系统修复计划

## 问题分析

### 问题1：战斗回合逻辑混乱
**当前问题：**
- 在 `Battle.tsx` 的 `executeEnemyTurn` 函数中，使用 `forEach` + `setTimeout` 的方式让敌人依次行动
- 每个敌人行动后会触发状态更新，但由于状态更新是异步的，可能导致回合混乱
- 敌人行动完毕后，使用 `setTimeout` 切换回玩家回合并增加回合数
- 问题：主角行动一次后，怪物可能行动了好几个回合

**根本原因：**
- 敌人回合的状态管理不够严格，没有确保所有敌人行动完毕后才切换回合
- `executeEnemyTurn` 函数中的异步操作可能导致状态不同步

### 问题2：经验值未真实增加，无升级逻辑
**当前问题：**
- 在 `App.tsx` 的 `handleBattleEnd` 函数中，只显示了获得的经验值（第977行）
- 没有将经验值增加到角色的 `exp` 字段
- 没有升级检查和升级逻辑

### 问题3：战斗结束后HP/MP被重置
**当前问题：**
- 在 `battleAdapter.ts` 的 `characterToBattleCharacter` 函数中（第116行），每次战斗开始时都会将 `currentMp` 设置为 `maxMp`
- 战斗结束后，角色的HP/MP状态没有保存回角色数据
- 没有在新一天到来和升级时恢复HP/MP的逻辑

## 修复方案

### 修复1：战斗回合逻辑优化

**修改文件：** `src/components/battle/Battle.tsx`

**修改内容：**
1. 重构 `executeEnemyTurn` 函数，使用更严格的状态管理
2. 确保所有敌人行动完毕后才切换回玩家回合
3. 使用 `async/await` 或 Promise 链来管理敌人行动的顺序
4. 在敌人行动过程中，禁止玩家操作

**具体实现：**
```typescript
// 修改 executeEnemyTurn 函数
const executeEnemyTurn = useCallback(() => {
  const aliveEnemies = battleState.enemies.filter(enemy => enemy.currentHp > 0);
  
  if (aliveEnemies.length === 0) return;
  
  // 使用递归或Promise链确保敌人依次行动
  let currentIndex = 0;
  
  const executeNextEnemy = () => {
    if (currentIndex >= aliveEnemies.length) {
      // 所有敌人行动完毕，切换回玩家回合
      setBattleState(prev => {
        if (prev.battleResult !== 'in_progress') {
          return prev;
        }
        
        // 移除过期的增益效果
        const updatedPlayer = removeExpiredBuffs(prev.player);
        const updatedEnemies = prev.enemies.map(enemy => removeExpiredBuffs(enemy));
        
        return {
          ...prev,
          player: updatedPlayer,
          enemies: updatedEnemies,
          isPlayerTurn: true,
          round: prev.round + 1
        };
      });
      return;
    }
    
    const enemy = aliveEnemies[currentIndex];
    currentIndex++;
    
    // 检查战斗是否已经结束
    setBattleState(currentState => {
      if (currentState.battleResult !== 'in_progress') {
        return currentState;
      }
      
      // 获取当前敌人状态
      const currentEnemy = currentState.enemies.find(e => e.id === enemy.id);
      if (!currentEnemy || currentEnemy.currentHp <= 0) {
        // 敌人已死亡，执行下一个敌人
        setTimeout(executeNextEnemy, 100);
        return currentState;
      }
      
      // 选择技能并执行攻击
      const availableSkills = currentEnemy.skills.filter(s => s.isAvailable);
      let selectedSkill = availableSkills[0];
      
      if (availableSkills.length > 1) {
        const aoeSkill = availableSkills.find(s => s.attackType === 'aoe');
        if (aoeSkill && currentEnemy.currentMp >= aoeSkill.mpCost) {
          selectedSkill = aoeSkill;
        } else {
          const randomIndex = Math.floor(Math.random() * availableSkills.length);
          selectedSkill = availableSkills[randomIndex];
        }
      }
      
      // 执行攻击（这里需要修改为同步更新状态）
      executeSkillAttack(currentEnemy, currentState.player, selectedSkill);
      
      return currentState;
    });
    
    // 延迟执行下一个敌人
    setTimeout(executeNextEnemy, 1000);
  };
  
  // 开始执行第一个敌人
  setTimeout(executeNextEnemy, 500);
}, [battleState.enemies]);
```

```typescript
/**
 * 检查并处理角色升级
 * @param currentExp 当前经验值
 * @param maxExp 升级所需经验值
 * @param currentLevel 当前等级
 * @returns 升级后的角色数据
 */
const checkAndLevelUp = useCallback((
  currentExp: number,
  maxExp: number,
  currentLevel: number
): { newExp: number; newMaxExp: number; newLevel: number; leveledUp: boolean } => {
  let exp = currentExp;
  let level = currentLevel;
  let leveledUp = false;
  
  // 检查是否可以升级（经验值溢出）
  while (exp >= maxExp && level < 125) { // 最高等级125
    exp -= maxExp;
    level++;
    leveledUp = true;
    
    // 计算下一级所需经验（可根据实际需求调整公式）
    maxExp = Math.floor(1000000 * Math.pow(1.1, level - 1));
  }
  
  return { newExp: exp, newMaxExp: maxExp, newLevel: level, leveledUp };
}, []);
```

```typescript
const handleBattleEnd = (result: BattleResult) => {
  setInBattle(false);
  setBattleParams(null);
  
  consumeTime(3);
  
  if (result === 'player_win' && currentBattleInteractableId) {

    if (battleParams?.enemiesData && battleParams.enemiesData.length > 0) {

          isBoss,
          character.luck
        );
      });
      
      const totalLoot = mergeLootResults(lootResults);
      
      // 添加金币
      if (totalLoot.gold > 0) {
        setPlayerResources(prev => ({
          ...prev,
          gold: prev.gold + totalLoot.gold
        }));
      }
      
      // 添加物品
      if (totalLoot.items.length > 0) {
        setInventory(prev => {
          const newInventory = [...prev];
          for (const lootItem of totalLoot.items) {
            const existingIndex = newInventory.findIndex(item => 
              item.id === lootItem.id && 
              (item.stackable === true || (item.maxStack && item.maxStack > 1))
            );
            
            if (existingIndex !== -1) {
              newInventory[existingIndex] = {
                ...newInventory[existingIndex],
                quantity: newInventory[existingIndex].quantity + lootItem.quantity,
              };
            } else {
              newInventory.push(lootItem);
            }
          }
          return newInventory;
        });
      }
      
      // 增加经验值并检查升级
      setCharacter(prev => {
        const newExp = prev.exp + totalLoot.experience;
        const { newExp: finalExp, newMaxExp, newLevel, leveledUp } = checkAndLevelUp(
          newExp,
          prev.maxExp,
          prev.level
        );
        
        // 如果升级了，恢复HP和MP到最大值，并更新属性
        if (leveledUp) {
          const baseAttrs = calculateCharacterBaseAttributes(newLevel);
          return {
            ...prev,
            level: newLevel,
            exp: finalExp,
            maxExp: newMaxExp,
            maxHp: baseAttrs.maxHp,
            currentHp: baseAttrs.maxHp, // 升级恢复HP
            maxStamina: baseAttrs.maxStamina,
            currentStamina: baseAttrs.maxStamina, // 升级恢复体力
            attackMin: baseAttrs.attackMin,
            attackMax: baseAttrs.attackMax,
            defense: baseAttrs.defense,
          };
        }
        
        return {
          ...prev,
          exp: finalExp,
          maxExp: newMaxExp,
        };
      });
      
      const lootMessages = [
        '战斗胜利！你击败了所有敌人。',
        `获得金币: ${totalLoot.gold}`,
        `获得经验: ${totalLoot.experience}`,
        ...totalLoot.messages.filter(msg => !msg.includes('获得金币') && !msg.includes('获得经验'))
      ];
      setInteractionLog(prev => [...prev, ...lootMessages]);
    } else {
      setInteractionLog(prev => [...prev, '战斗胜利！你击败了所有敌人。']);
    }
  } else {
    setInteractionLog(prev => [...prev, '战斗失败...下次再来挑战吧。']);
  }
  
  setCurrentBattleInteractableId(null);
};
```

#### 3.3 战斗开始时使用当前HP/MP

```typescript
export function characterToBattleCharacter(
  characterData: CharacterData,
  skills: SkillDetail[],
  isPlayer: boolean,
  gridPosition: GridPosition
): BattleCharacter {
  const maxMp = 100 + characterData.level * 10;
  
  const battleSkills: BattleSkill[] = skills
    .filter(skill => skill.isLearned)
    .map(skill => skillToBattleSkill(skill, maxMp, characterData.currentStamina));
  
  const battleCharacter: BattleCharacter = {
    id: characterData.id,
    name: characterData.playerName,
    level: characterData.level,
    maxHp: characterData.maxHp,
    currentHp: characterData.currentHp, // 使用当前HP
    maxMp: maxMp,
    currentMp: maxMp, // MP每次战斗恢复满（根据游戏设计）
    maxStamina: characterData.maxStamina,
    currentStamina: characterData.currentStamina, // 使用当前体力
    attackMin: characterData.attackMin,
    attackMax: characterData.attackMax,
    defense: characterData.defense,
    combatPower: 0,
    dodgeRate: characterData.dodgeRate,
    luck: characterData.luck,
    skills: battleSkills,
    buffs: [],
    isPlayer: isPlayer,
    gridPosition: gridPosition
  };
  
  battleCharacter.combatPower = calculateCombatPower(battleCharacter);
  
  return battleCharacter;
}
```
